class ElyzaLocalLLM(LLM):
    model_path: str
    chat_format: str = "llama-3"
    n_gpu_layers: int = 40
    n_ctx: int = 2048
    use_mmap: bool = False
    use_mlock: bool = False
    _llama: Llama = PrivateAttr()

    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self._llama = Llama(
            model_path=self.model_path,
            chat_format=self.chat_format,
            n_gpu_layers=self.n_gpu_layers,
            n_ctx=self.n_ctx,
            use_mmap=self.use_mmap,
            use_mlock=self.use_mlock,
        )

    def _call(self, prompt: str, stop=None):
        response = self._llama(prompt, max_tokens=1024, stop=stop)
        output_text = response["choices"][0]["text"]
        print(f"生成されたトークン数: {len(output_text.split())}")
        return output_text

    @property
    def _identifying_params(self):
        return {"model_path": self.model_path}

    @property
    def _llm_type(self):
        return "elyza_local"

# Elyzaモデルを設定
elyza_model = ElyzaLocalLLM(
    model_path="/model/Llama-3-ELYZA-JP-8B-q4_k_m.gguf",
    use_mlock=False,
    n_ctx=2048,
    temperature=0.5,
    top_k=40,
    top_p=0.9
)

# SerpAPIを設定
serp = SerpAPIWrapper()

# プロンプトテンプレートを作成
prompt_template = PromptTemplate(
    input_variables=["filtered_data", "search_results"],
    template=(
        "以下の鉄道データ（平常運転以外）および検索結果に基づいて、詳細かつ分かりやすい情報を提供してください:\n\n"
        "鉄道データ:\n{filtered_data}\n\n"
        "検索結果:\n{search_results}"
    )
)

# LLMチェーンを作成
chain = LLMChain(llm=elyza_model, prompt=prompt_template)

def fetch_and_filter_train_data():
    url = "https://ntool.online/data/train_all.json"
    response = requests.get(url)
    if response.status_code == 200:
        try:
            train_data = response.json()  # JSONデータを取得
            filtered_data = []

            # ネストされた構造に対応
            for key, rail_list in train_data.get("data", {}).items():
                for entry in rail_list:
                    # "平常運転" 以外をフィルタリング
                    rail_name = entry.get("railName", "不明")
                    status = entry.get("status", "不明")
                    info = entry.get("info", "なし")
                    if status != "平常運転":
                        filtered_data.append(f"鉄道名: {rail_name} / 状態: {status} / 詳細: {info}")

            return filtered_data
        except ValueError as e:
            raise Exception(f"JSONデータの解析に失敗しました: {e}")
    else:
        raise Exception(f"鉄道データを取得できませんでした。ステータスコード: {response.status_code}")

def main():
    try:
        # 鉄道データを取得してフィルタリング
        filtered_data = fetch_and_filter_train_data()
        if not filtered_data:
            print("すべての鉄道が平常運転のため、処理を終了します。")
            return

        # フィルタリングされたデータを文字列に変換
        filtered_data_str = "\n".join(filtered_data)

        # 検索結果を取得
        search_query = "鉄道 遅延 最新情報"
        search_results = serp.run(search_query)

        # 検索結果を短縮
        search_results_str = search_results[:300]

        # LangChainで処理
        result = chain.run(filtered_data=filtered_data_str, search_results=search_results_str)
        print("ELYZAの回答:")
        print(result)
    except Exception as e:
        print(f"エラーが発生しました: {e}")

if __name__ == "__main__":
    main()

import asyncio
from browser_use import Browser, Agent
from langchain.chat_models import ChatOpenAI

async def login(browser):
    """ Zabbix にログインする """
    await browser.goto("http://10.179.23.172:8080")

    # ユーザー名を入力
    await browser.type("input[name='name']", "Admin")

    # パスワードを入力
    await browser.type("input[name='password']", "zabbix")

    # ログインボタンをクリック
    await browser.click("button[type='submit']")

    # ダッシュボードがロードされるのを待機
    await browser.wait_for_selector("nav")

    print("ログイン完了")

async def navigate_to_hosts(browser):
    """ Data collection → Hosts へ移動し、Create host 画面を開く """
    await browser.click("a[title='Data collection']")
    await browser.click("a[title='Hosts']")
    await browser.click("button:has-text('Create host')")

    await browser.wait_for_selector("input[name='host']")
    print("ホスト作成画面へ移動完了")

async def configure_host(browser):
    """ ホスト情報を設定する """
    await browser.type("input[name='host']", "172.20.0.6")
    await browser.click("button:has-text('Select')")  # Host groups 選択

    # "Linux servers" を選択
    await browser.click("tr:has-text('Linux servers')")

    # 選択ウィンドウを閉じる
    await browser.click("button:has-text('Select')")

    print("ホスト情報設定完了")

async def add_snmp_interface(browser):
    """ SNMP インターフェースを追加する """
    await browser.click("button:has-text('Add')")  # インターフェースの追加
    await browser.click("input[value='SNMP']")  # SNMP を選択

    # "IP address" に "172.20.0.6" を入力
    await browser.type("input[name='ip']", "172.20.0.6")

    # "Add" ボタンをクリック
    await browser.click("button:has-text('Add')")

    print("SNMP インターフェース追加完了")

async def scenario_task():
    """ Agent を使ってシーケンシャルにタスクを実行 """
    async with Browser(headless=False) as browser:  # headless=False でブラウザを可視化
        await login(browser)              # ログイン
        await navigate_to_hosts(browser)  # ホスト作成ページへ移動
        await configure_host(browser)     # ホスト情報を設定
        await add_snmp_interface(browser) # SNMP インターフェース追加

        print("すべてのタスクが完了しました。")

async def main():
    """ LLM Agent を使用して `browser-use` のタスクを実行 """
    llm = ChatOpenAI(model="gpt-4o")  # OpenAI の GPT-4o を使用

    result = await Agent(
        task=scenario_task,
        llm=llm,
    ).run()

    print("Agent の実行結果:", result)

# 実行
asyncio.run(main())
